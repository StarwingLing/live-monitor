## 共享库两张表(DDL 属主本项目,bili db.py 锁步副本)

```sql
-- 房间级(总览页开关+齿轮弹窗写)
CREATE TABLE qq_notify (
    room_id INTEGER PRIMARY KEY,        -- 对应 rooms.room_id
    enabled INTEGER DEFAULT 0,          -- 通知开关
    group_ids TEXT DEFAULT '[]',        -- 目标群 JSON 数组
    template_on/off TEXT DEFAULT '',    -- 房间专属文案(空=用全局)
    updated_at TEXT);

-- 全局单行(管理页「开播账号管理」写)
CREATE TABLE qq_notify_global (
    id=1 固定单行, qq_account(QQ号预留), onebot_url/token(NapCat 连接),
    webui_url/token(扫码登录用), notify_on/off(全局开关),
    template_on/off(全局文案,默认自带 {cover}/{duration} 变量),
    debounce_offline(下播防抖秒,默认300), updated_at);
    -- notify_cover 列已停用(封面改纯模板变量),兼容保留不读
```
